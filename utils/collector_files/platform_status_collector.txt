script_variables

   FileIO platformFile = FileIO();
   string platformFileName = "platform_status.csv";
   bool platformFileOpened = false;

   int PLATFORM_EVENT_ID = 0;
   
   Array<string> platformColumnNames = {
      "Event_ID",
      "ISODate",
      "HMSTime",
      "SimulationTime",
      "Platform_Name",
      "Platform_Type",
      "Location_X",
      "Location_Y",
      "Location_Z"
   }; 
end_script_variables

execute at_interval_of 1 sec
   if (platformFileOpened)
   {
      for (int i = 0; i < WsfSimulation.PlatformCount(); i+=1)
      {
         WsfPlatform p = WsfSimulation.PlatformEntry(i);
         Array<string> pRow = {
            (string)PLATFORM_EVENT_ID,
            WsfDateTime.CurrentTime().ToStringIso8601(),
            WsfDateTime.CurrentTime().ToStringTime(),
            (string)(int)TIME_NOW,
            p.Name(),
            p.Type(),
            (string)p.X(),
            (string)p.Y(),
            (string)p.Z()
         };      
         platformFile.Writeln(",".Join(pRow));
         PLATFORM_EVENT_ID += 1;
      }
   }
end_execute
