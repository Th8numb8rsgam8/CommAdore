script void SetupParameters()
    
   commFileOpened = commFile.Open(commFileName, "out");
   if (commFileOpened)
   {
      commFile.Writeln(",".Join(columnNames));
   }
   else
   {
      writeln("FAILED TO OPEN " + commFileName);
   }

   trackFileOpened = trackFile.Open(trackFileName, "out");
   if (trackFileOpened)
   {
      trackFile.Writeln(",".Join(trackColumnNames));
   }
   else
   {
      writeln("FAILED TO OPEN " + trackFileName);
   }
   
   platformFileOpened = platformFile.Open(platformFileName, "out");
   if (platformFileOpened)
   {
      platformFile.Writeln(",".Join(platformColumnNames));
   }
   else
   {
      writeln("FAILED TO OPEN " + platformFileName);
   }

end_script

script void FinishVisualization()
    commFile.Close();
    trackFile.Close();
    platformFile.Close();
end_script
