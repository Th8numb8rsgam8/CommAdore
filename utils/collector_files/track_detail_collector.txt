script void track_LocalTrackCorrelation(WsfPlatform aPlatform, WsfTrackId aLocalTrackId, WsfTrackId aNonLocalTrackId)

   WsfLocalTrackList trackList = aPlatform.MasterTrackList();
   WsfLocalTrack localTrack;
   string localTracks = "";
   string rawTracks = "";
   
   foreach (WsfLocalTrack track in trackList)
   {
      localTracks += track.TrackId().ToString() + " ";
      if (track.TrackId() == aLocalTrackId)
      {
         localTrack = track;         
         for (int i = 0; i < track.RawTrackCount(); i+=1)
         {
            rawTracks += track.RawTrackId(i).ToString() + " ";
         }
      }
   }
   
   Array<string> dataRow = {
      (string)TRACK_EVENT_ID,
      WsfDateTime.CurrentTime().ToStringIso8601(),
      WsfDateTime.CurrentTime().ToStringTime(),
      (string)(int)TIME_NOW,
      (string)localTrack.TimeSinceStarted(),
      (string)localTrack.TimeSinceUpdated(),
      "LOCAL_TRACK_CORRELATION",
      aLocalTrackId.ToString(),
      aPlatform.Name(),
      aPlatform.Type(),
      aNonLocalTrackId.ToString(),
      (string)trackList.Count(),
      (string)localTrack.RawTrackCount(),
      localTracks,
      rawTracks,
      (string)localTrack.AltitudeKnown(),
      (string)localTrack.IsStale(),
      (string)aPlatform.Location().X(),
      (string)aPlatform.Location().Y(),
      (string)aPlatform.Location().Z(),
      (string)localTrack.CurrentLocation().X(),
      (string)localTrack.CurrentLocation().Y(),
      (string)localTrack.CurrentLocation().Z()
   };

   if (trackFileOpened)
   {
      trackFile.Writeln(",".Join(dataRow));
   }

   TRACK_EVENT_ID += 1;
end_script
